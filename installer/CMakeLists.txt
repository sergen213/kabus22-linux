cmake_minimum_required(VERSION 3.20)
project(kabus22_setup LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
find_package(LibArchive REQUIRED)
find_package(Threads REQUIRED)
add_library(kabus22_setup_core setup.cpp media.cpp)
target_include_directories(kabus22_setup_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(kabus22_setup_core PUBLIC PkgConfig::GTK3 PkgConfig::AV LibArchive::LibArchive Threads::Threads)
target_compile_options(kabus22_setup_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(kabus22-setup main.cpp)
target_link_libraries(kabus22-setup PRIVATE kabus22_setup_core)
set_target_properties(kabus22-setup PROPERTIES BUILD_RPATH_USE_ORIGIN TRUE INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS kabus22-setup RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(check_setup check_setup.cpp)
  target_link_libraries(check_setup PRIVATE kabus22_setup_core)
  add_test(NAME installer_security_boundaries COMMAND check_setup)
endif()
