cmake_minimum_required(VERSION 3.20)
project(kabus22_setup VERSION 0.4.7 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_definitions(K22_VERSION="${PROJECT_VERSION}")
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
find_package(LibArchive REQUIRED)
find_package(Threads REQUIRED)
add_library(kabus22_layout STATIC layout.cpp)
target_include_directories(kabus22_layout PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(kabus22_layout PRIVATE -Wall -Wextra -Wpedantic)
add_library(kabus22_setup_core setup.cpp media.cpp)
target_include_directories(kabus22_setup_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(kabus22_setup_core PUBLIC PkgConfig::GTK3 PkgConfig::AV LibArchive::LibArchive Threads::Threads kabus22_layout nlohmann_json::nlohmann_json)
target_compile_options(kabus22_setup_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(kabus22-setup main.cpp)
target_link_libraries(kabus22-setup PRIVATE kabus22_setup_core)
add_executable(kabus22-launcher launcher.cpp launcher_settings.cpp)
target_link_libraries(kabus22-launcher PRIVATE PkgConfig::GTK3 kabus22_layout)
target_compile_options(kabus22-launcher PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(kabus22-setup kabus22-launcher PROPERTIES BUILD_RPATH_USE_ORIGIN TRUE INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS kabus22-setup kabus22-launcher RUNTIME DESTINATION bin)
# Signed updater (Contract B). libcurl's OpenSSL backend already brings libcrypto,
# which also provides Ed25519 and SHA-256. Key: base64 raw 32-byte Ed25519 public key.
find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(nlohmann_json 3 REQUIRED)
set(K22_UPDATE_PUBLIC_KEY "2mfjUSeWRcSIPa4B3uaLUWxnBFPWulQYvdq1uv4PVVM=" CACHE STRING
  "Base64 raw 32-byte Ed25519 public key that verifies kabus22-update.json signatures")
if(NOT K22_UPDATE_PUBLIC_KEY MATCHES "^[A-Za-z0-9+/]+=?$")
  message(FATAL_ERROR "K22_UPDATE_PUBLIC_KEY must be base64 of a raw 32-byte Ed25519 key")
endif()
function(k22_update_library name)
  add_library(${name} STATIC update.cpp)
  target_link_libraries(${name} PUBLIC kabus22_layout CURL::libcurl OpenSSL::Crypto LibArchive::LibArchive nlohmann_json::nlohmann_json)
  target_compile_definitions(${name} PRIVATE K22_UPDATE_PUBLIC_KEY="${K22_UPDATE_PUBLIC_KEY}" ${ARGN})
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
endfunction()
k22_update_library(kabus22_update)
add_executable(kabus22-updater updater.cpp)
target_link_libraries(kabus22-updater PRIVATE kabus22_update)
target_compile_options(kabus22-updater PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(kabus22-updater PROPERTIES BUILD_RPATH_USE_ORIGIN TRUE INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS kabus22-updater RUNTIME DESTINATION bin)
